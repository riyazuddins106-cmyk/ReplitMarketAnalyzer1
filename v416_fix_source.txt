  1806: def numeric_similarity(
  1807:     a: float,
  1808:     b: float,
  1809:     scale: float,
  1810: ) -> float:
  1811: 
  1812:     return math.exp(
  1813:         -abs(a - b)
  1814:         / max(abs(scale), EPS)
  1815:     )
  1816: 
  1817: 
  1818: def path_row_similarity(
  1819:     current_row: Tuple[
  1820:         float,
  1821:         float,
  1822:         float,
  1823:         float,
  1824:     ],
  1825:     historical_row: Tuple[
  1826:         float,
  1827:         float,
  1828:         float,
  1829:         float,
  1830:     ],
  1831: ) -> float:
  1832: 
  1833:     (
  1834:         current_return,
  1835:         current_range,
  1836:         current_direction,
  1837:         current_body,
  1838:     ) = current_row
  1839: 
  1840:     (
  1841:         historical_return,
  1842:         historical_range,
  1843:         historical_direction,
  1844:         historical_body,
  1845:     ) = historical_row
  1846: 
  1847:     return_similarity = numeric_similarity(
  1848:         current_return,
  1849:         historical_return,
  1850:         1.0,
  1851:     )
  1852: 
  1853:     range_similarity = numeric_similarity(
  1854:         current_range,
  1855:         historical_range,
  1856:         1.0,
  1857:     )
  1858: 
  1859:     body_similarity = numeric_similarity(
  1860:         current_body,
  1861:         historical_body,
  1862:         1.0,
  1863:     )
  1864: 
  1865:     direction_similarity = (
  1866:         1.0
  1867:         if current_direction
  1868:         == historical_direction
  1869:         else 0.0
  1870:     )
  1871: 
  1872:     return (
  1873:         0.45 * return_similarity
  1874:         + 0.25 * range_similarity
  1875:         + 0.20 * body_similarity
  1876:         + 0.10 * direction_similarity
  1877:     )

  1880: def path_similarity(
  1881:     current: MarketState,
  1882:     record: ExperienceRecord,
  1883: ) -> float:
  1884: 
  1885:     values = [
  1886:         path_row_similarity(a, b)
  1887:         for a, b in zip(
  1888:             current.path_vector,
  1889:             record.path_vector,
  1890:         )
  1891:     ]
  1892: 
  1893:     return mean_or_zero(values)

  2001:         volatility=clamp(volatility),
