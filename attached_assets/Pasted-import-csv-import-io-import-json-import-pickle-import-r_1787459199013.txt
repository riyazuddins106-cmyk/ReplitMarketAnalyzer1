# ============================================================
# READ CSV
# ============================================================

def read_csv():

    print()
    print("=" * 80)
    print("READING XAU/USD CSV")
    print("=" * 80)

    text = CSV_FILE.read_text(
        encoding="utf-8-sig"
    )

    reader = csv.DictReader(
        io.StringIO(text)
    )

    if reader.fieldnames is None:

        raise RuntimeError(
            "CSV has no header."
        )

    print()
    print(
        "Columns:"
    )

    print(
        reader.fieldnames
    )

    required = {
        "datetime",
        "open",
        "high",
        "low",
        "close",
        "volume",
    }

    actual = set(
        reader.fieldnames
    )

    missing = (
        required - actual
    )

    if missing:

        raise RuntimeError(
            "Missing CSV columns: "
            f"{sorted(missing)}"
        )

    rows = list(reader)

    print(
        f"Rows read : "
        f"{len(rows):,}"
    )

    if not rows:

        raise RuntimeError(
            "CSV contains zero rows."
        )

    return rows


# ============================================================
# PARSE CANDLES
# ============================================================

def parse_candles(rows):

    candles = []

    for row_number, row in enumerate(
        rows,
        start=2
    ):

        try:

            dt_text = (
                row["datetime"]
                .strip()
            )

            dt = (
                datetime.fromisoformat(
                    dt_text.replace(
                        "Z",
                        "+00:00"
                    )
                )
            )

            if dt.tzinfo is None:

                dt = dt.replace(
                    tzinfo=timezone.utc
                )

            dt = dt.astimezone(
                timezone.utc
            )

            timestamp = int(
                dt.timestamp()
            )

            candle = {

                "instrument":
                    INSTRUMENT,

                "symbol":
                    SYMBOL,

                "timestamp":
                    timestamp,

                "datetime":
                    dt.isoformat(),

                "open":
                    float(row["open"]),

                "high":
                    float(row["high"]),

                "low":
                    float(row["low"]),

                "close":
                    float(row["close"]),

                "volume":
                    float(row["volume"]),
            }

            candles.append(
                candle
            )

        except Exception as error:

            raise RuntimeError(
                f"Invalid CSV row "
                f"{row_number}: "
                f"{error}"
            )

    return candles


# ============================================================
# OHLC VALIDATION
# ============================================================

def validate_ohlc(candle):

    o = candle["open"]
    h = candle["high"]
    l = candle["low"]
    c = candle["close"]

    if o <= 0:
        raise RuntimeError(
            "OPEN <= 0"
        )

    if h <= 0:
        raise RuntimeError(
            "HIGH <= 0"
        )

    if l <= 0:
        raise RuntimeError(
            "LOW <= 0"
        )

    if c <= 0:
        raise RuntimeError(
            "CLOSE <= 0"
        )

    if h < l:
        raise RuntimeError(
            "HIGH < LOW"
        )

    if h < o:
        raise RuntimeError(
            "HIGH < OPEN"
        )

    if h < c:
        raise RuntimeError(
            "HIGH < CLOSE"
        )

    if l > o:
        raise RuntimeError(
            "LOW > OPEN"
        )

    if l > c:
        raise RuntimeError(
            "LOW > CLOSE"
        )


# ============================================================
# CORPUS VALIDATION
# ============================================================

def validate_corpus(candles):

    print()
    print("=" * 80)
    print("CORPUS VALIDATION")
    print("=" * 80)

    if not candles:

        raise RuntimeError(
            "No candles."
        )

    duplicates = []

    out_of_order = []

    seen = set()

    previous = None

    for index, candle in enumerate(
        candles
    ):

        validate_ohlc(
            candle
        )

        timestamp = (
            candle["timestamp"]
        )

        if timestamp in seen:

            duplicates.append(
                timestamp
            )

        seen.add(
            timestamp
        )

        if (
            previous is not None
            and timestamp <= previous
        ):

            out_of_order.append(
                (
                    previous,
                    timestamp
                )
            )

        previous = timestamp

    print()
    print(
        f"Candles checked      : "
        f"{len(candles):,}"
    )

    print(
        f"Duplicates            : "
        f"{len(duplicates)}"
    )

    print(
        f"Out of order          : "
        f"{len(out_of_order)}"
    )

    if duplicates:

        raise RuntimeError(
            "Duplicate timestamps detected."
        )

    if out_of_order:

        raise RuntimeError(
            "Out-of-order candles detected."
        )

    print()
    print(
        "OHLC validation       : PASS"
    )

    print(
        "Duplicate validation  : PASS"
    )

    print(
        "Chronology validation : PASS"
    )


# ============================================================
# 5-MINUTE INTERVAL AUDIT
# ============================================================

def audit_intervals(candles):

    print()
    print("=" * 80)
    print("5-MINUTE INTERVAL AUDIT")
    print("=" * 80)

    expected = 300

    normal = 0

    gaps = []

    for i in range(
        1,
        len(candles)
    ):

        previous = (
            candles[i - 1]
            ["timestamp"]
        )

        current = (
            candles[i]
            ["timestamp"]
        )

        difference = (
            current - previous
        )

        if difference == expected:

            normal += 1

        elif difference > expected:

            missing = (
                difference
                // expected
            ) - 1

            gaps.append(
                {
                    "previous":
                        candles[i - 1]
                        ["datetime"],

                    "current":
                        candles[i]
                        ["datetime"],

                    "difference_seconds":
                        difference,

                    "missing_5m_bars":
                        missing,
                }
            )

        elif difference < expected:

            raise RuntimeError(
                "Timestamp interval "
                "less than 5 minutes."
            )

    print()
    print(
        f"Expected interval : "
        f"{expected} seconds"
    )

    print(
        f"Normal intervals  : "
        f"{normal:,}"
    )

    print(
        f"Gaps              : "
        f"{len(gaps):,}"
    )

    if gaps:

        print()
        print(
            "Gaps detected in supplied data."
        )

        print(
            "NO candles will be manufactured."
        )

        print()
        print(
            "First 10 gaps:"
        )

        for gap in gaps[:10]:

            print(
                f"{gap['previous']} -> "
                f"{gap['current']} | "
                f"missing="
                f"{gap['missing_5m_bars']}"
            )

    return {
        "expected_seconds":
            expected,

        "normal_intervals":
            normal,

        "gap_count":
            len(gaps),

        "gaps":
            gaps,
    }


# ============================================================
# CANDLE ANALYSIS
# ============================================================

def analyse_candle(candle):

    o = candle["open"]
    h = candle["high"]
    l = candle["low"]
    c = candle["close"]

    candle_range = (
        h - l
    )

    body = abs(
        c - o
    )

    upper_wick = (
        h - max(o, c)
    )

    lower_wick = (
        min(o, c) - l
    )

    if candle_range > 0:

        body_ratio = (
            body
            / candle_range
        )

        upper_ratio = (
            upper_wick
            / candle_range
        )

        lower_ratio = (
            lower_wick
            / candle_range
        )

    else:

        body_ratio = 0.0
        upper_ratio = 0.0
        lower_ratio = 0.0

    if c > o:

        direction = "bullish"

    elif c < o:

        direction = "bearish"

    else:

        direction = "neutral"

    if body_ratio < 0.10:

        candle_type = "doji_like"

    elif body_ratio > 0.70:

        candle_type = "strong_body"

    elif lower_ratio > 0.50:

        candle_type = "lower_rejection"

    elif upper_ratio > 0.50:

        candle_type = "upper_rejection"

    else:

        candle_type = "normal"

    result = dict(
        candle
    )

    result.update(
        {
            "direction":
                direction,

            "candle_type":
                candle_type,

            "range":
                candle_range,

            "body":
                body,

            "upper_wick":
                upper_wick,

            "lower_wick":
                lower_wick,

            "body_to_range":
                body_ratio,

            "upper_wick_to_range":
                upper_ratio,

            "lower_wick_to_range":
                lower_ratio,
        }
    )

    return result
