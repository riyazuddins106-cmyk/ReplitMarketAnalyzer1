    print("=" * 70)

    for k in NEIGHBORS:

        scored = []

        for record in records:

            distance = numeric_distance(
                current_features,
                record["features"]
            )

            scored.append(
                (
                    distance,
                    record["outcome"]
                )
            )

        scored.sort(
            key=lambda x: x[0]
        )

        neighbors = scored[:k]

        outcomes = [
            x[1]
            for x in neighbors
        ]

        counts = Counter(
            outcomes
        )

        directional = [
            x
            for x in outcomes
            if x in ("BUY", "SELL")
        ]

        print()
        print(
            f"TOP {k} HISTORICAL NEIGHBORS"
        )

        print(
            f"Bullish outcomes: "
            f"{counts['BUY']}"
        )

        print(
            f"Neutral outcomes: "
            f"{counts['NEUTRAL']}"
        )

        print(
            f"Bearish outcomes: "
            f"{counts['SELL']}"
        )

        if not directional:

            print(
                "Diagnostic prediction: NO TRADE"
            )

            print(
                "Confidence: 0.00%"
            )

            continue

        buy = counts["BUY"]
        sell = counts["SELL"]

        if buy > sell:

            prediction = "BUY"
            confidence = safe_div(
                buy,
                len(directional)
            )

        elif sell > buy:

            prediction = "SELL"
            confidence = safe_div(
                sell,
                len(directional)
            )

        else:

            prediction = "NO TRADE"
            confidence = 0.50

        print(
            f"Diagnostic prediction: "
            f"{prediction}"
        )

        print(
            f"Confidence: "
            f"{confidence:.2%}"
        )
