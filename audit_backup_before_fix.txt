def sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            block = f.read(1024 * 1024)
            if not block:
                break
            h.update(block)
    return h.hexdigest()


def safe_repr(value, limit=300):
    try:
        text = repr(value)
    except Exception:
        text = "<repr failed>"

    if len(text) > limit:
        return text[:limit] + "..."
    return text

def function_nodes():
    return [
        n for n in ast.walk(tree)
        if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))
    ]


def class_nodes():
    return [
        n for n in ast.walk(tree)
        if isinstance(n, ast.ClassDef)
    ]


def function_for_line(line):
    matches = []

    for fn in function_nodes():
        end = getattr(fn, "end_lineno", fn.lineno)

        if fn.lineno <= line <= end:
            matches.append(fn)

    if not matches:
        return None

    return max(matches, key=lambda x: x.lineno)


def function_name_for_line(line):
    fn = function_for_line(line)
    return fn.name if fn else "<module>"

def is_future_arithmetic(node):
    text = ast.get_source_segment(source, node)

    if not text:
        return False

    lower = text.replace(" ", "").lower()

    patterns = (
        "index+",
        "query_index+",
        "target+",
        "i+",
        "j+",
        "horizon",
    )

    return any(p in lower for p in patterns)


def future_reference_scan():
    hits = []

    for node in ast.walk(tree):

        if isinstance(node, ast.Subscript):
            text = ast.get_source_segment(source, node)

            if text and (
                "index +" in text
                or "index+" in text
                or "target" in text
                or "query_index +" in text
            ):
                hits.append(
                    (
                        node.lineno,
                        "SUBSCRIPT",
                        text.strip(),
                        function_name_for_line(node.lineno),
                    )
                )

        elif isinstance(node, ast.Slice):
            text = ast.get_source_segment(source, node)

            if text and (
                "index +" in text
                or "index+" in text
                or "target" in text
                or "query_index +" in text
            ):
                hits.append(
                    (
                        node.lineno,
                        "SLICE",
                        text.strip(),
                        function_name_for_line(node.lineno),
                    )
                )

        elif isinstance(node, ast.BinOp):
            if is_future_arithmetic(node):
                text = ast.get_source_segment(source, node)

                hits.append(
                    (
                        node.lineno,
                        "ARITHMETIC",
                        text.strip() if text else "",
                        function_name_for_line(node.lineno),
                    )
                )

    unique = []
    seen = set()

    for hit in sorted(hits):
        key = hit

        if key not in seen:
            seen.add(key)
            unique.append(hit)

    return unique

def resolve_argument(name, values):
    lower = name.lower()

    aliases = {
        "candles": (
            "candles",
            "data",
            "rows",
            "ohlc",
            "bars",
            "history",
        ),
        "index": (
            "index",
            "idx",
            "i",
            "query_index",
        ),
        "query_index": (
            "query_index",
            "index",
            "idx",
            "i",
        ),
        "horizon": (
            "horizon",
            "h",
        ),
        "atr": (
            "atr",
            "atr_values",
            "atr_series",
        ),
        "swings": (
            "swings",
            "swing_list",
        ),
        "states": (
            "states",
            "market_states",
        ),
        "experiences": (
            "experiences",
            "records",
            "experience_records",
        ),
        "window": (
            "window",
            "wf_window",
            "walk_forward_window",
        ),
    }

    if name in values:
        return values[name]

    for key, names in aliases.items():
        if lower in names and key in values:
            return values[key]

    return None


def invoke_adaptive(fn, values, required_only=True):
    """
    Invoke a function using its real runtime signature.

    Returns:
        (success, value, description)
    """

    try:
        sig = inspect.signature(fn)
    except Exception as exc:
        return False, None, f"signature unavailable: {exc}"

    kwargs = {}
    positional = []

    for param in sig.parameters.values():

        if param.kind in (
            inspect.Parameter.VAR_POSITIONAL,
            inspect.Parameter.VAR_KEYWORD,
        ):
            continue

        value = resolve_argument(param.name, values)

        if value is not None:
            if param.kind == inspect.Parameter.POSITIONAL_ONLY:
                positional.append(value)
            else:
                kwargs[param.name] = value

        elif param.default is inspect.Parameter.empty:

            return (
                False,
                None,
                f"cannot resolve required parameter '{param.name}' "
                f"for {fn.__name__}{sig}",
            )

    try:
        value = fn(*positional, **kwargs)
        return True, value, f"{fn.__name__}{sig}"

    except Exception as exc:
        return (
            False,
            None,
            f"{fn.__name__}{sig} raised "
            f"{type(exc).__name__}: {exc}",
        )

def mutate_suffix(data, start):

    altered = copy.deepcopy(data)

    for i in range(start, len(altered)):

        c = altered[i]

        # Modify OHLC values substantially while preserving
        # object type and timestamp.

        for attr in ("open", "high", "low", "close"):

            if hasattr(c, attr):

                old = getattr(c, attr)

                try:
                    setattr(c, attr, float(old) * 1.137 + 17.0)
                except Exception:
                    pass

        if hasattr(c, "volume"):

            try:
                setattr(
                    c,
                    "volume",
                    float(getattr(c, "volume")) * 2.0,
                )
            except Exception:
                pass

    return altered


def comparable_value(value):

    if is_dataclass(value):
        try:
            return asdict(value)
        except Exception:
            pass

    if isinstance(value, (list, tuple)):

        return [
            comparable_value(v)
            for v in value
        ]

    if isinstance(value, dict):

        return {
            str(k): comparable_value(v)
            for k, v in sorted(
                value.items(),
                key=lambda x: str(x[0]),
            )
        }

    if hasattr(value, "__dict__"):

        result_dict = {}

        for k, v in vars(value).items():

            if k.startswith("_"):
                continue

            result_dict[k] = comparable_value(v)

        return result_dict

    return value


def prefix_slice(value, n):

    if isinstance(value, (list, tuple)):
        return value[:n]

    return value
